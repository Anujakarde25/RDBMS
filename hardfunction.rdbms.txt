create table cp(cno int references client(cno),
pno int references Policy(pno));

Create OR Replace Function policy_fun(pamt int,mamt int) return int
as
     ans int;
Begin

     ans:=mamt-pamt;
     return ans;

END;
/

DECLARE

    a policy.pamt%type;
    b policy.mamt%type;
    c int;
    ucno client.cno%type;
BEGIN
      
      ucno:=&ucno;
     
     select pamt,mamt into a,b FROM Client,Policy,cp Where Client.cno=cp.cno and Policy.pno=cp.pno and client.cno=ucno;

  c:=policy_fun(a,b);

  DBMS_OUTPUT.PUT_LINE('Total maturityamt='|| c);

END;
